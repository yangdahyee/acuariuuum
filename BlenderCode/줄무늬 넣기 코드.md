### 줄무늬 넣기 코드
```python
# worldplanar_pink_stripes_fixed.py — Blender 4.x
import bpy

MAT_NAME   = "Body"          # 줄무늬 넣을 머티리얼
EMPTY_NAME = "StripeGizmo"   # 기준 Empty (이걸 움직이면 시작점 슬라이드)

# 색/파라미터
BASE_COL   = "#F7CBD0"       # 바탕 분홍
STRIPE_COL = "#E7686C"       # 줄무늬 분홍
AXIS       = 'Y'             # 'X'|'Y'|'Z' — 이 축에 수직인 평면 띠가 생김
SPACING    = 0.5            # 띠 간격(월드 단위). 값 ↑ = 띠 사이 멀어짐
WIDTH      = 0.5            # 띠 두께(0~1, 비율) 0.45면 주기의 45%
PHASE      = 0.2         # 시작 위치(0~1). Empty 이동으로도 조절 가능

def hexrgba(h, a=1.0):
    h = h.strip().lstrip('#')
    return (int(h[0:2],16)/255, int(h[2:4],16)/255, int(h[4:6],16)/255, a)

def ensure_empty(name=EMPTY_NAME):
    ob = bpy.data.objects.get(name)
    if not ob:
        ob = bpy.data.objects.new(name, None)
        bpy.context.scene.collection.objects.link(ob)
        ob.empty_display_type = 'ARROWS'
        ob.location = (0,0,0)
    return ob

def apply_planar_stripes():
    ref = ensure_empty()

    mat = bpy.data.materials.get(MAT_NAME) or bpy.data.materials.new(MAT_NAME)
    mat.use_nodes = True
    nt = mat.node_tree
    for n in list(nt.nodes): nt.nodes.remove(n)

    out  = nt.nodes.new("ShaderNodeOutputMaterial"); out.location  = (720, 0)
    bsdf = nt.nodes.new("ShaderNodeBsdfPrincipled"); bsdf.location = (480, 0)
    nt.links.new(bsdf.outputs["BSDF"], out.inputs["Surface"])
    bsdf.inputs["Base Color"].default_value = hexrgba(BASE_COL)

    tex  = nt.nodes.new("ShaderNodeTexCoord");       tex.location  = (-820, 0); tex.object = ref
    mapn = nt.nodes.new("ShaderNodeMapping");        mapn.location = (-640, 0)
    sep  = nt.nodes.new("ShaderNodeSeparateXYZ");    sep.location  = (-460, 0)
    div  = nt.nodes.new("ShaderNodeMath");           div.location  = (-280, 0); div.operation = 'DIVIDE'
    add  = nt.nodes.new("ShaderNodeMath");           add.location  = (-120, 0); add.operation = 'ADD'
    frac = nt.nodes.new("ShaderNodeMath");           frac.location = (  40, 0); frac.operation = 'FRACT'   # <-- 고침!
    ramp = nt.nodes.new("ShaderNodeValToRGB");       ramp.location = ( 200, 0); ramp.color_ramp.interpolation = 'CONSTANT'

    # 파라미터
    div.inputs[1].default_value  = max(SPACING, 1e-6)   # x / SPACING
    add.inputs[1].default_value  = PHASE                # + phase (0~1)
    e0, e1 = ramp.color_ramp.elements[0], ramp.color_ramp.elements[1]
    e0.position = 0.0
    e1.position = max(0.0, min(1.0, WIDTH))             # 0~WIDTH 구간 = 줄무늬
    e0.color = hexrgba(STRIPE_COL); e1.color = hexrgba(BASE_COL)

    # 축 선택(Separate XYZ 출력 인덱스)
    idx = {'X':0, 'Y':1, 'Z':2}[AXIS]

    # 링크
    nt.links.new(tex.outputs['Object'], mapn.inputs['Vector'])
    nt.links.new(mapn.outputs['Vector'], sep.inputs['Vector'])
    nt.links.new(sep.outputs[idx],       div.inputs[0])
    nt.links.new(div.outputs['Value'],   add.inputs[0])
    nt.links.new(add.outputs['Value'],   frac.inputs[0])
    nt.links.new(frac.outputs['Value'],  ramp.inputs['Fac'])
    nt.links.new(ramp.outputs['Color'],  bsdf.inputs['Base Color'])

    print(f"[OK] Planar stripes on {MAT_NAME}. Move '{EMPTY_NAME}' along {AXIS} to slide start.")

apply_planar_stripes()
```
